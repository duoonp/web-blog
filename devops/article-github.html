Controlled GitHub deploys without Enterprise subscription

The Problem

Deploying directly to a private Kubernetes cluster (like GKE) from standard GitHub Actions runners is a challenge. You either have to expose your cluster's API endpoint to the public internet, or pay for a GitHub Enterprise subscription to get advanced private networking features. Neither is ideal for a secure, budget-conscious setup.

The Fix

You can bypass this by splitting your CI/CD pipeline into two parts. The standard GitHub-hosted runners handle the heavy lifting of building and pushing the Docker image to your registry. Then, a self-hosted runner (gke-private-runner) located inside your private network picks up the deployment job. Because this runner is already inside the firewall, it can run kubectl commands securely against your private cluster.

The Workflow Code

Below is the GitHub Actions workflow that implements this pattern. Notice how the deploy-prod job specifically targets the private runner.

name: Build, Push & Deploy

on:
  release:
    types: [published]
  workflow_dispatch:
    inputs:
      deploy_prod:
        description: 'Deploy to Prod?'
        type: boolean
        default: true

env:
  PROJECT_ID: ${{ secrets.GCP_PROJECT_ID }}
  REGION: europe-west4
  REPO_NAME: my-repo
  IMAGE_NAME: my-service
  REGISTRY: europe-west4-docker.pkg.dev
  DEPLOY_NAME_PROD: my-service

jobs:
  check-build-push:
    runs-on: ubuntu-latest
    steps:
      - name: Authenticate to Google Cloud
        uses: google-github-actions/auth@v2
        with:
          credentials_json: ${{ secrets.GCP_SA_KEY }}

      - name: Log in to Artifact Registry
        uses: docker/login-action@v3
        with:
          registry: ${{ env.REGISTRY }}
          username: _json_key
          password: ${{ secrets.GCP_SA_KEY }}

      - name: Check if Image Exists
        id: check_image
        run: |
          if docker manifest inspect ${{ env.REGISTRY }}/${{ env.PROJECT_ID }}/${{ env.REPO_NAME }}/${{ env.IMAGE_NAME }}:${{ github.sha }} > /dev/null 2>&1; then
            echo "exists=true" >> $GITHUB_OUTPUT
            echo "Image already exists! Skipping build."
          else
            echo "exists=false" >> $GITHUB_OUTPUT
            echo "Image does not exist. Proceeding with build."
          fi

      - uses: actions/checkout@v4
        if: steps.check_image.outputs.exists == 'false'

      - name: Free Disk Space (Ubuntu)
        if: steps.check_image.outputs.exists == 'false'
        uses: jlumbroso/free-disk-space@main
        with:
          tool-cache: false
          android: true
          dotnet: true
          haskell: true
          large-packages: false
          docker-images: false
          swap-storage: false

      - uses: docker/setup-buildx-action@v3
        if: steps.check_image.outputs.exists == 'false'

      - name: Sanitize ref name for Docker tag
        if: steps.check_image.outputs.exists == 'false'
        id: sanitize
        run: echo "SAFE_REF=$(echo '${{ github.ref_name }}' | sed 's/[^a-zA-Z0-9.-]/-/g')" >> $GITHUB_ENV

      - name: Build and push (with registry cache)
        if: steps.check_image.outputs.exists == 'false'
        uses: docker/build-push-action@v6
        with:
          context: .
          push: true
          tags: |
            ${{ env.REGISTRY }}/${{ env.PROJECT_ID }}/${{ env.REPO_NAME }}/${{ env.IMAGE_NAME }}:latest
            ${{ env.REGISTRY }}/${{ env.PROJECT_ID }}/${{ env.REPO_NAME }}/${{ env.IMAGE_NAME }}:${{ github.sha }}
            ${{ env.REGISTRY }}/${{ env.PROJECT_ID }}/${{ env.REPO_NAME }}/${{ env.IMAGE_NAME }}:${{ env.SAFE_REF }}
          cache-from: type=registry,ref=${{ env.REGISTRY }}/${{ env.PROJECT_ID }}/${{ env.REPO_NAME }}/${{ env.IMAGE_NAME }}:buildcache
          cache-to: type=registry,ref=${{ env.REGISTRY }}/${{ env.PROJECT_ID }}/${{ env.REPO_NAME }}/${{ env.IMAGE_NAME }}:buildcache,mode=max

  deploy-prod:
    if: ${{ github.event.inputs.deploy_prod == 'true' }}
    needs: check-build-push
    runs-on: gke-private-runner
    environment: prod
    steps:
      - name: Install kubectl
        uses: azure/setup-kubectl@v4

      - name: Sanitize ref name for Docker tag
        id: sanitize
        run: echo "SAFE_REF=$(echo '${{ github.ref_name }}' | sed 's/[^a-zA-Z0-9.-]/-/g')" >> $GITHUB_ENV

      - name: Update prod deployment image and start backend
        run: |
          echo -n 'SAFE_REF: ${{ env.SAFE_REF }} \n'
          kubectl set image deployment/${{ env.DEPLOY_NAME_PROD }} ${{ env.DEPLOY_NAME_PROD }}=${{ env.REGISTRY }}/${{ env.PROJECT_ID }}/${{ env.REPO_NAME }}/${{ env.IMAGE_NAME }}:${{ env.SAFE_REF }} -n backend

          echo -n 'Scaling down... \n'
          kubectl scale deploy ${{ env.DEPLOY_NAME_PROD }} -n backend --replicas=0
          kubectl delete po -l app="${{ env.DEPLOY_NAME_PROD }}" -n backend

          echo -n 'Waiting for scale down... \n'
          while true; do POD=$(kubectl get pod -l app="${{ env.DEPLOY_NAME_PROD }}" -n backend --no-headers 2>/dev/null | wc -l); if [ "$POD" -eq 0 ]; then break; fi;  sleep 1; done

          echo -n 'Scaling up... \n'
          kubectl scale deploy "${{ env.DEPLOY_NAME_PROD }}" -n backend --replicas=1
          kubectl rollout status deployment ${{ env.DEPLOY_NAME_PROD }} -n backend --timeout=10m

      - name: Start backend and verify health
        run: |
          echo "Waiting for backend to start..."
          until kubectl exec deploy/${{ env.DEPLOY_NAME_PROD }} -c ${{ env.DEPLOY_NAME_PROD }} -n backend -- python3 -c "import urllib.request; assert urllib.request.urlopen('http://localhost:8080/login').getcode() == 200" &>/dev/null; do
            echo "Waiting for 200 OK..."
            sleep 2
          done
          echo "${{ env.DEPLOY_NAME_PROD }} backend is ready!"
